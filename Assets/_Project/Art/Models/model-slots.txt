#   Ashtray_814_Case = Slots/ashtray.fbx
#   Chair_814        = Slots/chair.fbx | rotY=90
#
# سه تغییردهنده، همه اختیاری، با | جدا می‌شوند:
#
#   rotY=90     چرخش حول محور عمودی، درجه
#   stretch=1   کشیدن در سه محور جدا (پیش‌فرض: بزرگ‌کردن یکنواخت)
#   lift=0.05   بالا بردن، متر
#
# قاعده: پیش‌فرض یکنواخت بزرگ می‌کند تا مدل کشیده نشود؛ اندازه را از جعبه‌ای
# می‌گیرد که آن آبجکت الان اشغال کرده، پس چیدمان صحنه به‌هم نمی‌خورد.
#
